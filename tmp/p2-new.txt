/**
 * Hapus permanen invoice beserta efek stok & kas-nya (dipakai purgeInvoice).
 * Riwayat stok milik invoice + entri kas dihapus, saldo kas dihitung ulang,
 * lalu dokumen invoice dihapus dari database.
 */
async function hardDeleteInvoice(ctx: any, idInvoice: string) {
  const inv = await findOneByKey(ctx, "invoice", "idInvoice", idInvoice);
  if (!inv) return false;
  const allHist = await ctx.db.query("stokHistory").collect();
  const hist = allHist.filter(
    (h: any) =>
      h.keterangan === `Invoice ${idInvoice}` ||
      (h.keterangan ?? "").includes(`(${idInvoice})`),
  );
  for (const h of hist) await ctx.db.delete(h._id);
  const kas = await findOneByKey(ctx, "kas", "id", `INV-${idInvoice}`);
  if (kas) await ctx.db.delete(kas._id);
  await recomputeKas(ctx);
  await ctx.db.delete(inv._id);
  return true;
}

/**
 * Hapus invoice → PINDAH KE SAMPAH (soft delete). Kas & stok TIDAK diubah
 * selama invoice masih di Sampah, sehingga invoice bisa dipulihkan utuh lewat
 * restoreInvoice. Efek dibatalkan hanya saat dihapus permanen (purgeInvoice).
 */
export const deleteInvoice = mutation({
  args: { idInvoice: v.string() },
  handler: async (ctx, { idInvoice }) => {
    logRequest("deleteInvoice", { idInvoice });
    const inv = await findOneByKey(ctx, "invoice", "idInvoice", idInvoice);
    if (!inv) return { deleted: false };
    if (!(inv as any).deletedAt) {
      await ctx.db.patch(inv._id, { deletedAt: Date.now() } as any);
    }
    logResponse("deleteInvoice", { deleted: true, trashed: true });
    return { deleted: true, trashed: true };
  },
});

/**
 * Pulihkan invoice dari Sampah kembali ke daftar aktif. Karena efek kas &
 * stok tidak pernah dibatalkan saat soft-delete, invoice kembali UTUH tanpa
 * perlu menghitung ulang apa pun.
 */
export const restoreInvoice = mutation({
  args: { idInvoice: v.string() },
  handler: async (ctx, { idInvoice }) => {
    logRequest("restoreInvoice", { idInvoice });
    const inv = await findOneByKey(ctx, "invoice", "idInvoice", idInvoice);
    if (!inv) return { restored: false };
    if ((inv as any).deletedAt) {
      await ctx.db.patch(inv._id, { deletedAt: undefined } as any);
    }
    logResponse("restoreInvoice", { restored: true });
    return { restored: true };
  },
});

/**
 * Hapus permanen invoice dari Sampah: batalkan efek stok & kas (riwayat stok
 * dan entri kas dihapus, saldo dihitung ulang) lalu hapus dokumen. Tidak bisa
 * dipulihkan lagi.
 */
export const purgeInvoice = mutation({
  args: { idInvoice: v.string() },
  handler: async (ctx, { idInvoice }) => {
    logRequest("purgeInvoice", { idInvoice });
    const ok = await hardDeleteInvoice(ctx, idInvoice);
    logResponse("purgeInvoice", { deleted: ok });
    return { deleted: ok };
  },
});

/** Simpan langganan push (endpoint + kunci) — upsert agar tidak duplikat. */
export const savePushSubscription = mutation({
  args: {
    endpoint: v.string(),
    keys: v.object({ p256dh: v.string(), auth: v.string() }),
  },
  handler: async (ctx, { endpoint, keys }) => {
    if (!endpoint || !endpoint.startsWith("https://")) {
      return badRequest("Endpoint push tidak valid", { endpoint });
    }
    const existing = await (ctx.db.query("pushSubscription") as any)
      .filter((q: any) => q.eq(q.field("endpoint"), endpoint))
      .first();
    if (existing) {
      await ctx.db.patch(existing._id, { keys, createdAt: Date.now() } as any);
    } else {
      await ctx.db.insert("pushSubscription", { endpoint, keys, createdAt: Date.now() } as any);
    }
    return { saved: true };
  },
});

/** Hapus langganan push (dipakai saat user menonaktifkan notifikasi). */
export const removePushSubscription = mutation({
  args: { endpoint: v.string() },
  handler: async (ctx, { endpoint }) => {
    const existing = await (ctx.db.query("pushSubscription") as any)
      .filter((q: any) => q.eq(q.field("endpoint"), endpoint))
      .first();
    if (existing) await ctx.db.delete(existing._id);
    return { removed: !!existing };
  },
});
